; MCM Extender - Stentorious

; Default requirements check
string_var sMessage = ""
int iRequirements = 1

; xNVSE
float fVersion = 0
float fRequired = 6.48
if eval (fVersion = GetNVSEVersionFull) < fRequired
	MessageBoxExAlt ({int iTemp} => 1) "^Requirements Checker^Outdated xNVSE detected!%r%rInstalled: %g.%.1f%rRequired: %g.%.1f+|Ok" (Flr fVersion) ((Fmod fVersion 1) * 10) (Flr fRequired) ((Fmod fRequired 1) * 10)
	iRequirements = 0
endif

; xNVSE plugins
array_var aRequirements = ar_Map "JohnnyGuitarNVSE"::528 "ShowOffNVSE Plugin"::184 "UI Organizer Plugin"::230 "MCM Extensions"::1
array_var aForE = Ar_Null
if eval (Ar_Size aRequirements) > 0
	foreach aForE <- aRequirements
		fVersion = GetPluginVersion (aForE["key"])
		fRequired = *aForE
		if eval fVersion < fRequired
			sMessage = Sv_Construct "%z%r> %z: %g -> %g+" sMessage (aForE["key"]) fVersion fRequired
		endif
	loop
	if eval sMessage != ""
		MessageBoxExAlt ({int iTemp} => 1) "^Requirements Checker^Outdated xNVSE plugins detected!%r%rPlease update the following plugins:%r%z|Ok" sMessage
		iRequirements = 0
	endif
endif

; Mod files
if eval IsModLoaded "The Mod Configuration Menu.esp" == 0
	MessageBoxExAlt ({int iTemp} => 1) "^Requirements Checker^Required mods files missing!%r%rPlease install the following mod files:%r> The Mod Configuration Menu.esp|Ok"
	iRequirements = 0
endif

aRequirements = aForE = Ar_Null
sv_Destruct sMessage

if eval iRequirements == 0
	return
endif

Call (CompileScript "MCMExtender\OnRestart.gek")
